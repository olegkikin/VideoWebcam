cmake_minimum_required(VERSION 3.24)
project(VideoWebcam VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build on Windows with Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
set(BASE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/baseclasses")
if(NOT EXISTS "${BASE}/streams.h")
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 first.")
endif()
file(GLOB BASE_SOURCES "${BASE}/*.cpp")
list(FILTER BASE_SOURCES EXCLUDE REGEX "(dllentry|dllsetup)\\.cpp$")
add_library(dshow_base STATIC ${BASE_SOURCES})
target_include_directories(dshow_base PUBLIC "${BASE}")
target_compile_definitions(dshow_base PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NON_CONFORMING_SWPRINTFS)
target_compile_options(dshow_base PRIVATE /W0 /Zc:strictStrings- /wd4996)
target_link_libraries(dshow_base PUBLIC strmiids winmm ole32 oleaut32 uuid advapi32)
add_library(frame_bus STATIC src/frame_bus.cpp)
target_include_directories(frame_bus PUBLIC src)
target_compile_definitions(frame_bus PUBLIC NOMINMAX)
target_link_libraries(frame_bus PUBLIC advapi32)
add_library(VideoWebcamFilter SHARED src/filter.cpp src/filter.def "${BASE}/dllentry.cpp")
target_link_libraries(VideoWebcamFilter PRIVATE dshow_base frame_bus)
target_compile_options(VideoWebcamFilter PRIVATE /W4 /EHsc)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set_target_properties(VideoWebcamFilter PROPERTIES OUTPUT_NAME VideoWebcamFilter64)
  add_executable(VideoWebcam WIN32 src/app.cpp src/decoder.cpp src/app.rc)
  target_link_libraries(VideoWebcam PRIVATE frame_bus comctl32 comdlg32 shell32 gdi32 user32 ole32)
  target_compile_options(VideoWebcam PRIVATE /W4 /EHsc /utf-8)
  target_link_options(VideoWebcam PRIVATE /MANIFEST:NO)
endif()
enable_testing()
add_executable(camera_tests tests/camera_tests.cpp)
target_link_libraries(camera_tests PRIVATE dshow_base frame_bus)
target_compile_options(camera_tests PRIVATE /W4 /EHsc)
add_test(NAME camera_pipeline COMMAND camera_tests "$<TARGET_FILE:VideoWebcamFilter>")
set_tests_properties(camera_pipeline PROPERTIES TIMEOUT 90)
if(TARGET VideoWebcam)
  add_executable(decoder_tests tests/decoder_tests.cpp src/decoder.cpp)
  target_link_libraries(decoder_tests PRIVATE frame_bus)
  target_compile_options(decoder_tests PRIVATE /W4 /EHsc)
endif()
